/*
Definition for Node
class Node {
public:
    int data;
    Node* left;
    Node* right;

    Node(int val) {
        data = val;
        left = right = nullptr;
    }
};
*/

class Solution {
  public:
    vector<int> bottomView(Node *root) {
        // code here
        queue<pair<Node*,int>> q;
        map<int,int> mp;
        q.push({root,0});vector<int> ans;
        while(!q.empty()){
            auto it=q.front();
            Node* curr=it.first;
            int line=it.second;
            mp[line]=curr->data;
            q.pop();
            if(curr->left){
                q.push({curr->left,line-1});
            }
            if(curr->right){
                q.push({curr->right,line+1});
            }
        }
        for(auto it:mp){
            ans.push_back(it.second);
        }
        
        return ans;
        
    }
};
