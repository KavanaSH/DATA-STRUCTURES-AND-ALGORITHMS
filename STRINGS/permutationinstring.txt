class Solution {
public:
    bool isfreqsame(int freq1[],int freq2[]){
        for(int i=0;i<26;i++){
            if(freq1[i]!=freq2[i]){
                return false; 
            }
        }
        return true;
    }
    bool checkInclusion(string s1, string s2) {
        int freq1[26]={0};
        for(char ch:s1){
            freq1[ch-'a']++;
        }
        int winlen=s1.length();
        for(int i=0;i<s2.length();i++){
            int winidx=0;int idx=i;int freq2[26]={0};
            while(winidx<winlen && idx<s2.length()){
                freq2[s2[idx]-'a']++;
                idx++;
                winidx++;
            }
            if(isfreqsame(freq1,freq2)){
                return true;
            }
        }
        return false;

    }
};
